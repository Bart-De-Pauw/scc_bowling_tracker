# ============================================================================
# RASPBERRY PI DEPLOYMENT GUIDE
# ============================================================================

# PREREQUISITES:
# 1. Raspberry Pi 4 (4GB+ RAM) with Raspberry Pi OS
# 2. SSH access to your Pi
# 3. Docker and Docker Compose installed

# INSTALLATION STEPS:

# 1. Install Docker on Raspberry Pi:
ssh pi@your-pi-ip
curl -fsSL https://get.docker.com -o get-docker.sh
sudo sh get-docker.sh
sudo usermod -aG docker pi

# 2. Install Docker Compose:
sudo apt-get install -y docker-compose

# 3. Clone/Create project directory:
mkdir bowling-tracker
cd bowling-tracker

# 4. Copy the docker-compose.yml and project files into this directory
# Structure should be:
# bowling-tracker/
#   ├── docker-compose.yml
#   ├── .env
#   ├── backend/
#   │   ├── Dockerfile
#   │   ├── package.json
#   │   ├── server.js
#   │   └── database.js
#   └── frontend/
#       ├── Dockerfile
#       ├── nginx.conf
#       ├── package.json
#       └── src/

# 5. Create .env file:
cat > .env << EOF
EMAIL_USER=your-email@gmail.com
EMAIL_PASSWORD=your-app-password
EOF

# 6. Build and start containers:
docker-compose up -d

# 7. Check logs:
docker-compose logs -f

# 8. Access the application:
# Frontend: http://your-pi-ip:3000
# Backend: http://your-pi-ip:5000

# ============================================================================
# USEFUL DOCKER COMMANDS FOR RASPBERRY PI
# ============================================================================

# View running containers:
docker-compose ps

# View logs:
docker-compose logs backend
docker-compose logs frontend

# Stop services:
docker-compose down

# Restart services:
docker-compose restart

# Update and rebuild (after code changes):
docker-compose down
docker-compose up -d --build

# Access Pi database:
docker exec -it bowling-backend sh
sqlite3 /data/bowling.db

# ============================================================================
# BACKUP STRATEGY
# ============================================================================

# Backup database daily:
# Add to crontab: crontab -e
0 2 * * * cp /path/to/bowling-tracker/data/bowling.db /backup/bowling.db.$(date +\%Y\%m\%d)

# Keep backups for 30 days:
0 2 * * * find /backup -name "bowling.db.*" -mtime +30 -delete

# ============================================================================
# SYSTEMD SERVICE FOR AUTO-START (optional)
# ============================================================================
# Create: /etc/systemd/system/bowling-tracker.service

[Unit]
Description=Bowling Tracker Docker Compose
After=docker.service
Requires=docker.service

[Service]
Type=oneshot
User=pi
WorkingDirectory=/home/pi/bowling-tracker
ExecStart=/usr/bin/docker-compose up -d
ExecStop=/usr/bin/docker-compose down
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target

# Enable autostart:
sudo systemctl enable bowling-tracker.service
sudo systemctl start bowling-tracker.service
